Example: CI/CD screenshot capture script visual-test.sh #!/bin/bash
# visual-regression-check.sh
BASE_URL="https://staging.yourapp.com"
API_KEY="YOUR_SCREENSHOTAPI_KEY"
ENDPOINT="https://shot.screenshotapi.net/screenshot"
VIEWPORTS=("1280x800" "768x1024" "375x812")
PAGES=("/dashboard" "/checkout" "/pricing" "/login")
for PAGE in "${PAGES[@]}"; do
 for VIEWPORT in "${VIEWPORTS[@]}"; do
 WIDTH=$(echo $VIEWPORT | cut -d'x' -f1)
 HEIGHT=$(echo $VIEWPORT | cut -d'x' -f2)
 FILENAME="screenshots/${PAGE////}-${VIEWPORT}.png"
 curl -s "${ENDPOINT}?token=${API_KEY}&url=${BASE_URL}${PAGE}&width=${WIDTH}&height=${HEIGHT}&output=image&file_type=png&block_ads=true&incognito=true" -o "${FILENAME}"
 echo "Captured: ${FILENAME}"
 done
done
